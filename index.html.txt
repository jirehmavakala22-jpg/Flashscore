    <script>
        function openMatch(home, away, score, time) {
            document.getElementById('detail-teams').innerText = home + ' vs ' + away;
            document.getElementById('detail-score').innerText = score;
            document.getElementById('detail-status').innerText = 'En direct (' + time + ')';

            document.getElementById('page-home').classList.remove('active');
            document.getElementById('page-detail').classList.add('active');
        }

        function showHome() {
            document.getElementById('page-detail').classList.remove('active');
            document.getElementById('page-home').classList.add('active');
